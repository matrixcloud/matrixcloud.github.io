Blockquotes without a marker render like Hugo's default. */}} {{- if eq .Type "alert" -}} {{- $types := slice "note" "info" "tip" "warning" -}} {{- $aliases := dict "hint" "tip" "important" "tip" "success" "tip" "check" "tip" "done" "tip" "warn" "warning" "caution" "warning" "attention" "warning" "danger" "warning" "error" "warning" "failure" "warning" "fail" "warning" "missing" "warning" "bug" "warning" "todo" "note" "example" "note" "abstract" "note" "summary" "note" "tldr" "note" "question" "note" "help" "note" "faq" "note" "quote" "note" "cite" "note" -}} {{- $name := lower (printf "%s" .AlertType) -}} {{- $style := cond (in $types $name) $name (or (index $aliases $name) "note") -}} {{- $title := printf "%s" .AlertTitle -}} {{- if not $title -}} {{- $title = cond (in $types $name) (i18n $style) $name -}} {{- end -}}
